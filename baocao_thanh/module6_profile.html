525double _calculateShippingFee(String? city) {
526  if (city == null) return 50000;
527  final mienBac = ['Hà Nội', 'Hải Phòng', 'Quảng Ninh'...];
528  final mienNam = ['TP.HCM', 'Bình Dương', 'Đồng Nai'...];
529  if (mienNam.contains(city)) return 25000; // Gần kho
530  if (mienBac.contains(city)) return 35000;
531  return 50000; // Miền Trung/Tây Nguyên
532}

547// Card number formatter: 1234 5678 9012 3456
548class CardNumberFormatter extends TextInputFormatter {
549  TextEditingValue formatEditUpdate(old, new) {
550    var text = new.text.replaceAll(' ', '');
551    var formatted = '';
552    for (int i = 0; i < text.length; i++) {
553      if (i > 0 && i % 4 == 0) formatted += ' ';
554      formatted += text[i];
555    }
556    return TextEditingValue(text: formatted);
557  }
