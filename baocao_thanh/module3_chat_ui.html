33  Future<void> _fetchUserInfo() async {
34    if (_userId == null) return;
35    final doc = await FirebaseFirestore.instance
36        .collection('users').doc(_userId).get();
37    if (doc.exists) {
38      setState(() {
39        _userName = doc['fullName'] ?? doc['name'] ?? 'User';
40        _userEmail = doc['email'] ?? '';
41      });
42      // Create/update chat session
43      await _chatService.createOrUpdateChatSession(_userId!, _userName, _userEmail);
44      // Send product info if available
45      if (widget.product != null) {
46        _sendProductInfo();
47      }
48    }
49  }

55        '[IMG]${p.imageUrl}[/IMG]';
56    _chatService.sendUserMessage(_userId!, msg);
57  }

102  Widget _buildMessageBubble(ChatMessage msg) {
103    final isMe = !msg.isAdmin;
104    return Align(
105      alignment: isMe ? Alignment.centerRight : Alignment.centerLeft,
106      child: Bubble(

108        nip: isMe ? BubbleNip.rightBottom : BubbleNip.leftBottom,
109        color: isMe ? const Color(0xFFD29062) : Colors.grey[200],
110        child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
111          _parseMessageContent(msg.text, isMe),
112          if (msg.orderId != null) _buildOrderAttachment(msg),
113          Text(DateFormat('HH:mm').format(msg.timestamp),

120  Widget _parseMessageContent(String text, bool isMe) {
121    // Parse [IMG]url[/IMG] tags
122    final imgRegex = RegExp(r'\[IMG\](.*?)\[\/IMG\]');
123    final match = imgRegex.firstMatch(text);
124    if (match != null) {
125      final imgUrl = match.group(1)!;
126      final cleanText = text.replaceAll(imgRegex, '').trim();
127      return Column(children: [
128        ClipRRect(borderRadius: BorderRadius.circular(8), 
129          child: Image.network(imgUrl, width: 150, fit: BoxFit.cover)),
130        if (cleanText.isNotEmpty) Text(cleanText, style: TextStyle(color: isMe ? Colors.white : Colors.black)),
131      ]);
132    }
133    return Text(text, style: TextStyle(color: isMe ? Colors.white : Colors.black));
134  }
