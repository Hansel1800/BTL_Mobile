69  @override

71    super.initState();
72    _initializeAuthState();

74
75  void _initializeAuthState() {
76    _authSubscription = FirebaseAuth.instance.authStateChanges().listen((user) async {
77      if (!mounted) return;
78      setState(() {
79        _currentUser = user;
80      });
81      if (user != null) {
82        // Initialize Notifications for ALL users (User & Admin)
83        NotificationService().initialize();
84
85        final userDoc = await FirebaseFirestore.instance
86            .collection("users")
87            .doc(user.uid)
88            .get();
89        if (!mounted) return;
90        if (userDoc.exists) {
91          setState(() {
92            _userRole = userDoc['role'];
93          });
94        }
95      } else {
96        // User logged out
97        setState(() {
98            _userRole = null;
99        });
100      }
101    });
102  }
103
104  @override
105  void dispose() {
106    _authSubscription?.cancel();
107    super.dispose();
108  }
109
110  @override
111  Widget build(BuildContext context) {
112    debugPrint('AuthStateHandler build - currentUser: ${_currentUser?.uid}, role: $_userRole');
113    
114    if (_currentUser == null) {
115      debugPrint('Showing LoginScreen');
116      return const LoginScreen();
117    }
118    if (_userRole == null) {
119      debugPrint('Showing loading indicator');
120      return const Scaffold(
121        body: Center(
122          child: CircularProgressIndicator(),
123        ),
124      );
125    }
126    debugPrint('Showing ${_userRole == "Admin" ? "AdminScreen" : "UserMainScreen"}');
127    return _userRole == "Admin" ? const AdminScreen() : const UserMainScreen();
128  }
