173// Custom Painter for Mountains
174class MountainPainter extends CustomPainter {
175  final bool isNight;
176  MountainPainter({required this.isNight});
177
178  @override
179  void paint(Canvas canvas, Size size) {
180    final paint = Paint()..color = isNight ? const Color(0xFF2D3748) : const Color(0xFF4A5568);
181    final path = Path()
182      ..moveTo(0, size.height)
183      ..lineTo(size.width * 0.25, size.height * 0.4)
184      ..lineTo(size.width * 0.5, size.height * 0.7)
185      ..lineTo(size.width * 0.75, size.height * 0.2)
186      ..lineTo(size.width, size.height)
187      ..close();
188    canvas.drawPath(path, paint);
189  }
190
191  @override
192  bool shouldRepaint(MountainPainter old) => isNight != old.isNight;
193}

21  Future<void> _signup() async {
22    if (!_formKey.currentState!.validate()) return;
23    
24    // Validation
25    final name = _nameController.text.trim();
26    final phone = _phoneController.text.trim();
27    final email = _emailController.text.trim();
28    final password = _passwordController.text;
29
30    // Name validation: only letters and spaces
31    if (!RegExp(r'^[a-zA-ZÀ-ỹ\s]+$').hasMatch(name)) {
32      _showError('Tên chỉ được chứa chữ cái');
33      return;
34    }
35
36    // Phone validation: 10 digits, starts with 0
37    if (!RegExp(r'^0\d{9}$').hasMatch(phone)) {
38      _showError('SĐT phải có 10 số và bắt đầu bằng 0');
39      return;
40    }
41
42    // Email validation
43    if (!RegExp(r'^[\w-\.]+@([\w-]+\.)+[\w-]{2,4}$').hasMatch(email)) {
44      _showError('Email không hợp lệ');
45      return;
46    }
47
48    setState(() => _isLoading = true);
49    try {
50      // Call AuthService with role = "User"
51      final result = await _authService.signup(
52        name: name,
53        phone: phone,
54        email: email,
55        password: password,
56        role: 'User',
57      );

70      }
