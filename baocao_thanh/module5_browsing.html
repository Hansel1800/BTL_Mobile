66// Filter products
67List<Product> _filterProducts(List<Product> products) {
68  return products.where((p) {
69    // Search filter (diacritic-insensitive)
70    if (_searchQuery.isNotEmpty) {
71      final name = _removeDiacritics(p.name.toLowerCase());
72      final query = _removeDiacritics(_searchQuery.toLowerCase());
73      if (!name.contains(query)) return false;
74    }
75    // Size filter
76    if (_selectedSizes.isNotEmpty && !_selectedSizes.any((s) => p.sizes.contains(s))) return false;
77    // Color filter
78    if (_selectedColors.isNotEmpty && !_selectedColors.any((c) => p.colors.contains(c))) return false;
79    // Price range filter
80    if (p.price < _minPrice || p.price > _maxPrice) return false;
81    return true;
82  }).toList();
83}

621String _removeDiacritics(String str) {
